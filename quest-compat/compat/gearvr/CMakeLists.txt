cmake_minimum_required(VERSION 3.22)
project(hb_legacy_adapter LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT ANDROID)
  message(FATAL_ERROR "The legacy adapter requires the Android NDK; host tests do not compile it.")
endif()
if(NOT TARGET OpenXR::openxr_loader)
  find_package(OpenXR REQUIRED CONFIG)
endif()
add_library(hb_legacy_xr STATIC src/xr_backend.cpp)
set_target_properties(hb_legacy_xr PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(hb_legacy_xr PUBLIC src)
target_link_libraries(hb_legacy_xr PUBLIC OpenXR::openxr_loader EGL GLESv3 android log)
target_compile_options(hb_legacy_xr PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
option(HB_BUILD_LEGACY_VRAPI "Build the experimental VrApi frontend with separately supplied SDK headers" OFF)
if(HB_BUILD_LEGACY_VRAPI)
  set(VRAPI_ROOT "" CACHE PATH "VrApi SDK directory containing Include/VrApi.h")
  foreach(header VrApi.h VrApi_Helpers.h VrApi_Input.h VrApi_SystemUtils.h VrApi_Vulkan.h)
    if(NOT EXISTS "${VRAPI_ROOT}/Include/${header}")
      message(FATAL_ERROR "Missing ${header} under VRAPI_ROOT/Include; no ABI declarations will be fabricated.")
    endif()
  endforeach()
  add_library(hb_legacy_vrapi SHARED src/vrapi_system.cpp src/vrapi_frame.cpp src/vrapi_swapchain.cpp src/vrapi_input.cpp src/vrapi_driver.cpp)
  # Direct APK replacement must expose the SONAME requested by OVRPlugin.
  set_target_properties(hb_legacy_vrapi PROPERTIES OUTPUT_NAME vrapi)
  target_compile_definitions(hb_legacy_vrapi PRIVATE HB_DEFAULT_QUEST=1)
  target_include_directories(hb_legacy_vrapi PRIVATE "${VRAPI_ROOT}/Include")
  target_link_libraries(hb_legacy_vrapi PRIVATE hb_legacy_xr dl)
  target_compile_options(hb_legacy_vrapi PRIVATE -Wall -Wextra -Wno-missing-field-initializers)
  target_link_options(hb_legacy_vrapi PRIVATE -Wl,--no-undefined)
endif()
