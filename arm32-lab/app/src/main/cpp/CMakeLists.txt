cmake_minimum_required(VERSION 3.22)
project(nexa_arm32_lab LANGUAGES C CXX)
if(NOT EXISTS "${UNICORN_ROOT}/include/unicorn/unicorn.h")
    message(FATAL_ERROR "Set unicornRoot to the pinned Unicorn 2.1.4 source checkout")
endif()
set(UNICORN_ARCH arm CACHE STRING "" FORCE)
set(UNICORN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(UNICORN_INSTALL OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${UNICORN_ROOT}" "${CMAKE_BINARY_DIR}/unicorn")
add_library(nexa_arm32 SHARED probe.cpp)
target_include_directories(nexa_arm32 PRIVATE "${UNICORN_ROOT}/include")
target_link_libraries(nexa_arm32 PRIVATE unicorn log)
target_link_options(nexa_arm32 PRIVATE "-Wl,-z,max-page-size=16384")
